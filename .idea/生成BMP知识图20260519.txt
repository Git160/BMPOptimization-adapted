//====================================================
// 其余行政区（统一模板）
//====================================================
WITH [
'白樟镇','白中镇','坂东镇','池园镇','大洋镇',
'红星乡','金沙镇','梅城镇','梅溪镇','盘谷乡',
'三溪乡','上莲乡','省璜镇','塔庄镇','汤川乡',
'霞拔乡','云龙乡'
] AS towns

UNWIND towns AS town

MATCH (a:AdminDistrict {name:town}),
      (f1:ScenarioFactor {name:'土地利用'}),
      (f3:ScenarioFactor {name:'地形位置'}),
      (f4:ScenarioFactor {name:'治理对象'}),
      (f5:ScenarioFactor {name:'决策主体'})

CREATE (a)-[:HAS_FACTOR {value:'林地、农田、茶园'}]->(f1)
CREATE (a)-[:HAS_FACTOR {value:'坡地、洪泛区'}]->(f3)
CREATE (a)-[:HAS_FACTOR {value:'总氮'}]->(f4)
CREATE (a)-[:HAS_FACTOR {value:'乡镇政府'}]->(f5);
